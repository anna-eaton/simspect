\newcommand{\graphicon}[2]{%
  \begin{scope}[shift={#2}, scale=\icon]
    \foreach \o in {2,1} \draw[fill=white] ($(-0.27+0.06*\o,-0.13-0.06*\o)$) rectangle ++(0.42,0.32);
    \draw[fill=white] (-0.27,-0.13) rectangle (0.15,0.19);
    \foreach \p/\q in {a/{-0.18,0.1},b/{0.06,0.1},c/{-0.18,-0.05},d/{0.06,-0.05}} \fill (\q) circle (0.6pt) coordinate (#1\p);
    \draw[line width=0.3pt] (#1a) -- (#1b) -- (#1d) (#1a) -- (#1c) -- (#1d);
    \path (0,0.19) coordinate (#1-n) (0,-0.25) coordinate (#1-s);
  \end{scope}}
% document (#3 = fill, #4 = glyph or empty for text lines); anchors #1-n/-s/-e/-w
\newcommand{\docicon}[4]{%
  \begin{scope}[shift={#2}, scale=\icon]
    \draw[fill=#3] (-0.17,-0.22) -- (0.17,-0.22) -- (0.17,0.13) -- (0.08,0.22) -- (-0.17,0.22) -- cycle;
    \draw (0.08,0.22) -- (0.08,0.13) -- (0.17,0.13);
    \if\relax\detokenize{#4}\relax
      \foreach \y in {0.06,-0.03,-0.12} \draw[line width=0.3pt] (-0.1,\y) -- (0.1,\y);
    \else \node[font=\tiny] at (0,-0.04) {#4};\fi
    \path (0,0.22) coordinate (#1-n) (0,-0.22) coordinate (#1-s) (0.17,0) coordinate (#1-e) (-0.17,0) coordinate (#1-w);
  \end{scope}}
% small bug, drawn around (0,0)
\newcommand{\bugicon}[1]{%
  \begin{scope}[shift={#1}, scale=1.35, line width=0.35pt]
    \foreach \y in {0.035,0,-0.035} {\draw (-0.085,\y-0.012) -- (0.085,\y+0.012); \draw (-0.085,\y+0.012) -- (0.085,\y-0.012);}
    \fill (0,-0.005) ellipse (0.045 and 0.065);
    \fill (0,0.075) circle (0.028);
    \draw (-0.012,0.095) -- (-0.04,0.13) (0.012,0.095) -- (0.04,0.13);
  \end{scope}}
