% ============================ TUNING KNOBS ============================
\def\cyc{6.2}        % pt per simulated cycle (same scale in every axis segment)
\def\brk{9}          % pt width of each axis break
\def\rowh{17}        % pt between instruction rows
\def\barh{4}         % pt height of an instruction bar
\def\labx{-4}        % pt, right edge of the row labels
\def\winlw{0.9pt}    % line width of the violation-window frame
\def\mrk{1.7pt}      % radius of the footprint markers
\def\boxgap{9}       % pt between the time axis and the defense-specific note (note spans the full figure width)
\def\defshade{black!12} % fill marking defense-specific elements
% axis segments [start,end], in cycles after i0's fetch; breaks between them
\def\aS{-0.5}\def\aE{8.5}       % front end; lc retires at 8
\def\bS{39.5}\def\bE{41.5}      % transmitter issues at 40
\def\cS{68.5}\def\cE{71.5}      % L1D eviction at 70
\def\dS{99.5}\def\dE{102.5}     % speculative-buffer fill at 101
\def\eS{5006.5}\def\eE{5013.5}  % branch resolves 5007, squash 5008, fnc retires 5013
% ======================================================================

\pgfmathsetmacro\wA{(\aE-\aS)*\cyc}
\pgfmathsetmacro\wB{(\bE-\bS)*\cyc}
\pgfmathsetmacro\wC{(\cE-\cS)*\cyc}
\pgfmathsetmacro\wD{(\dE-\dS)*\cyc}
\pgfmathsetmacro\wE{(\eE-\eS)*\cyc}
% x position (pt) of cycle #1 inside segment a..e
\pgfmathdeclarefunction{xa}{1}{\pgfmathparse{(#1-\aS)*\cyc}}
\pgfmathdeclarefunction{xb}{1}{\pgfmathparse{\wA+\brk+(#1-\bS)*\cyc}}
\pgfmathdeclarefunction{xc}{1}{\pgfmathparse{\wA+\wB+2*\brk+(#1-\cS)*\cyc}}
\pgfmathdeclarefunction{xd}{1}{\pgfmathparse{\wA+\wB+\wC+3*\brk+(#1-\dS)*\cyc}}
\pgfmathdeclarefunction{xe}{1}{\pgfmathparse{\wA+\wB+\wC+\wD+4*\brk+(#1-\eS)*\cyc}}

\tikzset{
  every node/.style={font=\footnotesize, inner sep=0pt,
    execute at begin node={\hyphenpenalty=10000\exhyphenpenalty=10000}},
  stg/.style={font=\scriptsize, anchor=south},
  ann/.style={font=\scriptsize\itshape},
}

% bar edges (top and bottom lines) from x=#2 to x=#3 at row y=#1; #4 = extra draw options
\newcommand{\edges}[4]{\draw[#4] (#2,#1+\barh/2) -- (#3,#1+\barh/2) (#2,#1-\barh/2) -- (#3,#1-\barh/2);}
% vertical tick across a bar at x=#2, row y=#1
\newcommand{\tck}[2]{\draw (#2,#1-\barh/2) -- (#2,#1+\barh/2);}
% a bar that runs from x=#2 through every later segment up to segment e at x=#3
\newcommand{\longbar}[3]{%
  \edges{#1}{#2}{\wA}{}
  \edges{#1}{\wA}{\wA+\brk}{densely dotted}
  \edges{#1}{\wA+\brk}{\wA+\brk+\wB}{}
  \edges{#1}{\wA+\brk+\wB}{\wA+2*\brk+\wB}{densely dotted}
  \edges{#1}{\wA+2*\brk+\wB}{\wA+2*\brk+\wB+\wC}{}
  \edges{#1}{\wA+2*\brk+\wB+\wC}{\wA+3*\brk+\wB+\wC}{densely dotted}
  \edges{#1}{\wA+3*\brk+\wB+\wC}{\wA+3*\brk+\wB+\wC+\wD}{}
  \edges{#1}{\wA+3*\brk+\wB+\wC+\wD}{\wA+4*\brk+\wB+\wC+\wD}{densely dotted}
  \edges{#1}{\wA+4*\brk+\wB+\wC+\wD}{#3}{}}
