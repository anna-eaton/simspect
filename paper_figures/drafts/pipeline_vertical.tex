% one document page (#3 fill, #4 glyph or empty = text lines), centred at (#2)
\newcommand{\docpage}[4]{%
  \begin{scope}[shift={#2}]%
    \draw[fill=#3] (-0.17,-0.22) -- (0.17,-0.22) -- (0.17,0.13) -- (0.08,0.22) -- (-0.17,0.22) -- cycle;%
    \draw (0.08,0.22) -- (0.08,0.13) -- (0.17,0.13);%

    \else \node[font=\tiny] at (0,-0.04) {#4};\fi%
  \end{scope}}%
% single document icon; anchors #1-n/-s/-e/-w
